#   bicubic: Bicubic interpolation
#   gaussian: Gaussian smoothing
#   catmull_rom: Catmull-Rom cubic spline

#   lanczos: Lanczos reconstruction
#   spline36: Spline interpolation (36 samples)
#
# Recommendations:
#   * Anime: Hermite
#   * Live Action: Spline36
#
# [dehalo]
# What is a halo? A halo is the bright (or dark) outline that hugs the edges of
# objects, most visible around hair, text and thin lines. It is usually left
# behind by sharpening that was applied earlier: at the studio, by an upscale,
# or by an older encode. Halos look like a glow tracing the edge, and because
# they are high contrast detail the encoder has to spend bitrate on them.
# Removing them makes edges look cleaner and frees bitrate for real detail.
#
# Anime and older upscaled/sharpened material benefit most. Clean live action
# usually does not need this. Dehalo is a destructive filter: too much of it
# eats thin lines and flattens edges, so start gentle and only raise the
# strength if halos are still visible.
#
# Dehalo runs BEFORE denoise, so denoise smooths whatever dehalo leaves behind.
#
# dehalo=True/False
#   Master switch. False (default) skips the filter entirely.
#
# dehalo_rx / dehalo_ry (default 2.0, must be at least 1.0)
#   How wide a halo the filter looks for, in pixels, horizontally (rx) and
#   vertically (ry). Think of it as "how far out from the edge does the glow
#   reach?" Raise it for thick, obvious halos; lower it for thin ones.
#   Too high softens the picture and starts eating detail near edges.
#     1.5 = subtle halos
#     2.0 = typical (recommended starting point)
#     3.0 = thick halos on heavily sharpened sources
#   Keep rx and ry equal unless the halos are clearly stretched in one
#   direction (some anamorphic or badly resized sources).
#
# dehalo_brightstr (default 1.0, range 0.0-1.0)
#   Strength against BRIGHT halos - the white/light glow around dark lines.
#   This is the common case. 1.0 removes them fully, 0.5 halves them,
#   0.0 leaves them alone. Lower this first if lines start looking thin.
#
# dehalo_darkstr (default 0.0, range 0.0-1.0)
#   Strength against DARK halos - the dark smear on the light side of an edge.
#   Left at 0.0 on purpose: dark halos are rarer, and removing them very easily
#   destroys line art. Only raise it (try 0.2-0.5) if you can actually see
#   dark rings, and check the result.
#
# dehalo_lowsens / dehalo_highsens (default 50 / 50, range 0-100)
#   These control the mask that decides which pixels count as "halo". The
#   filter is applied fully below lowsens and skipped entirely above highsens,
#   blending in between. Lower values = more aggressive (treats more of the
#   frame as halo). Higher values = more conservative (only the most obvious
#   halos are touched). The defaults are fine for almost everything; raise
#   both to about 60-70 if dehalo is damaging areas that had no halos.
#
# dehalo_ss (default 1.5)
#   Supersampling factor. The filter internally works at a higher resolution
#   so the edges it rebuilds do not come out jagged. 1.5 is the usual value;
#   2.0 is slightly cleaner but slower; 1.0 disables it and is fastest but can
#   leave aliasing (stair-stepping) on diagonal lines.
#
# Suggested starting points:
#   * Anime with visible edge glow: dehalo=True, rx/ry=2.0, brightstr=1.0
#   * Mild halos / careful pass:    dehalo=True, rx/ry=1.5, brightstr=0.6
#   * Clean live action:            dehalo=False
#
# [denoise]
# .bat files can overwrite this setting. This is useful for 5fish where a light denoise
# is ideal due to the 5fish high level of detail retention
